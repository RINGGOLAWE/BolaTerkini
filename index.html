import { createClient } from "npm:@supabase/supabase-js@2"; const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const supabase = createClient( SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, ); const MAX_ARTICLES = 5; const RSS_FEEDS = [ { name: "BBC Sport", url: "https://feeds.bbci.co.uk/sport/football/rss.xml", }, { name: "ESPN", url: "https://www.espn.com/espn/rss/soccer/news", }, { name: "Sky Sports", url: "https://www.skysports.com/rss/12040", }, ]; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; type NewsItem = { title: string; description: string; link: string; image_url: string | null; source: string; pubDate: string; }; function jsonResponse( data: unknown, status = 200, ) { return new Response(JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, }); } // Membersihkan teks HTML dan karakter XML function cleanText(value: string): string { return value .replace(//g, "") .replace(//gi, "") .replace(//gi, "") .replace(/<[^>]*>/g, " ") .replace(/ /gi, " ") .replace(/&/gi, "&") .replace(/"/gi, '"') .replace(/'|'/gi, "'") .replace(/</gi, "<") .replace(/>/gi, ">") .replace(/&#(\d+);/g, (_, code) => String.fromCodePoint(Number(code)) ) .replace(/&#x([0-9a-f]+);/gi, (_, code) => String.fromCodePoint(parseInt(code, 16)) ) .replace(/\s+/g, " ") .trim(); } // Mengambil isi tag XML function getTag( xml: string, tagName: string, ): string { const escaped = tagName.replace(/[.*+?^${}()|[\]\\]/g, "\\$&"); const regex = new RegExp( `<${escaped}\\b[^>]*>([\\s\\S]*?)<\\/${escaped}\\s*>`, "i", ); const match = xml.match(regex); return match?.[1]?.trim() ?? ""; } // Mengambil URL gambar dari RSS jika tersedia function getImage(item: string): string | null { const media = item.match(/]*url=["']([^"']+)["']/i) || item.match(/]*url=["']([^"']+)["']/i) || item.match(/]*url=["']([^"']+)["']/i); if (media?.[1]) { return media[1]; } const description = getTag(item, "description"); const image = description.match( /]+src=["']([^"']+)["']/i, ); return image?.[1] ?? null; } // Mengambil berita dari satu RSS feed async function fetchFeed(feed: { name: string; url: string; }): Promise { try { const response = await fetch(feed.url, { headers: { "User-Agent": "BolaTerkini News Bot", }, signal: AbortSignal.timeout(15000), }); if (!response.ok) { console.error( `Gagal mengambil ${feed.name}: ${response.status}`, ); return []; } const xml = await response.text(); const items = xml.match( /]*>[\s\S]*?<\/item>/gi, ) ?? []; return items.map((item) => { const title = cleanText(getTag(item, "title")); const rawDescription = getTag(item, "description") || getTag(item, "content:encoded"); const description = cleanText(rawDescription); const link = cleanText(getTag(item, "link")); const pubDate = cleanText(getTag(item, "pubDate")); return { title, description, link, image_url: getImage(item), source: feed.name, pubDate, }; }).filter((item) => item.title.length > 0 && item.link.startsWith("http") ); } catch (error) { console.error(`Error feed ${feed.name}:`, error); return []; } } // Menerjemahkan teks ke Bahasa Indonesia async function translateToIndonesian( text: string, ): Promise { if (!text) return ""; try { const url = new URL( "https://translate.googleapis.com/translate_a/single", ); url.searchParams.set("client", "gtx"); url.searchParams.set("sl", "auto"); url.searchParams.set("tl", "id"); url.searchParams.set("dt", "t"); url.searchParams.set("q", text.slice(0, 4500)); const response = await fetch(url.toString(), { signal: AbortSignal.timeout(12000), }); if (!response.ok) { return text; } const result = await response.json(); const translated = (result[0] as unknown[]) .map((part: unknown) => { if (Array.isArray(part) && typeof part[0] === "string") { return part[0]; } return ""; }) .join(""); return translated.trim() || text; } catch (error) { console.error("Terjemahan gagal:", error); return text; } } // Menentukan kategori berita function getCategory(text: string): string { const value = text.toLowerCase(); if ( /transfer|signing|signs for|joins|loan deal|rumour|rumor/.test(value) ) { return "Transfer"; } if ( /premier league|la liga|serie a|bundesliga|ligue 1|league table/.test(value) ) { return "Liga"; } if ( /champions league|europa league|conference league|world cup|fa cup/.test(value) ) { return "Kompetisi"; } if ( /injury|injured|fitness|sidelined|return from injury/.test(value) ) { return "Cedera"; } if ( /manager|coach|sacked|appointed|head coach/.test(value) ) { return "Pelatih"; } if ( /national team|international|indonesia|garuda/.test(value) ) { return "Tim Nasional"; } return "Berita"; } // Memeriksa admin yang sedang login async function verifyAdmin(req: Request): Promise { const authorization = req.headers.get("Authorization"); if (!authorization?.startsWith("Bearer ")) { return false; } const token = authorization.replace("Bearer ", ""); const { data: userData, error: userError } = await supabase.auth.getUser(token); if (userError || !userData.user) { return false; } const { data: isAdmin, error: adminError } = await supabase.rpc("is_bolaterkini_admin"); if (adminError) { console.error("Pemeriksaan admin gagal:", adminError); return false; } return isAdmin === true; } // Proses utama mengambil berita async function fetchNews() { const allNews: NewsItem[] = []; // Mengambil berita dari semua sumber const feedResults = await Promise.all( RSS_FEEDS.map((feed) => fetchFeed(feed)), ); for (const news of feedResults) { allNews.push(...news); } // Mengurutkan berita berdasarkan tanggal terbaru allNews.sort((a, b) => { const dateA = new Date(a.pubDate).getTime() || 0; const dateB = new Date(b.pubDate).getTime() || 0; return dateB - dateA; }); // Menghapus duplikat dari hasil RSS const uniqueNews = Array.from( new Map( allNews.map((item) => [item.link, item]), ).values(), ); let saved = 0; let skipped = 0; const errors: string[] = []; for (const news of uniqueNews) { if (saved >= MAX_ARTICLES) break; try { // Cek apakah berita sudah pernah disimpan const { data: existing, error: checkError } = await supabase .from("articles") .select("id") .eq("source_url", news.link) .maybeSingle(); if (checkError) { throw checkError; } if (existing) { skipped++; continue; } // Terjemahkan judul dan ringkasan const translatedTitle = await translateToIndonesian( news.title, ); const translatedDescription = await translateToIndonesian( news.description, ); const category = getCategory( `${news.title} ${news.description}`, ); // RSS hanya menyediakan ringkasan, bukan artikel lengkap. const content = translatedDescription || "Ringkasan berita belum tersedia. Silakan buka tautan sumber asli untuk membaca informasi selengkapnya."; // Simpan sebagai draft const { error: insertError } = await supabase .from("articles") .insert({ title: translatedTitle, content, category, image_url: news.image_url, source: news.source, source_url: news.link, is_published: false, }); if (insertError) { throw insertError; } saved++; } catch (error) { console.error("Gagal menyimpan berita:", error); errors.push( `${news.source}: ${String(error)}`, ); } } return { success: true, message: "Proses pengambilan berita selesai.", total_found: uniqueNews.length, saved, skipped, errors, }; } // Handler Supabase Edge Function Deno.serve(async (req: Request) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders, }); } if (req.method !== "POST") { return jsonResponse({ success: false, message: "Gunakan metode POST.", }, 405); } try { // Pastikan hanya admin yang bisa menjalankan fungsi const isAdmin = await verifyAdmin(req); if (!isAdmin) { return jsonResponse({ success: false, message: "Akses ditolak. Hanya admin yang dapat mengambil berita.", }, 403); } const body = await req.json().catch(() => ({})); if (body.action !== "fetch_news") { return jsonResponse({ success: false, message: "Action tidak dikenal.", }, 400); } const result = await fetchNews(); return jsonResponse(result); } catch (error) { console.error("Kesalahan Edge Function:", error); return jsonResponse({ success: false, message: "Terjadi kesalahan saat mengambil berita.", error: String(error), }, 500); } });
